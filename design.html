import time
import json
import threading
import webview
def push_live_telemetry(window):
    """Background bridge thread pushing Python trading_engine state to GPU UI."""
    while True:
        time.sleep(0.4)
        # Replace with your actual check_state / trading_engine dictionary:
        state_payload = {
            "livePrice": 75841.84,
            "equity": 23494.27,
            "sessionPnl": 48.27,
            "basketPnl": 6.84,
            "conf": 91.0
        }
        try:
            window.evaluate_js(f"window.receivePythonState && window.receivePythonState({json.dumps(state_payload)})")
        except Exception:
            break
if __name__ == "__main__":
    html_path = os.path.abspath("nexus_dashboard.html")
    win = webview.create_window(
